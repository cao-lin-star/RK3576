cmake_minimum_required(VERSION 3.8)
project(rk3576_footbath_navigation)
find_package(ament_cmake REQUIRED)
set(CMAKE_CXX_STANDARD 17)
foreach(dep nav2_rotation_shim_controller nav2_core nav2_costmap_2d nav2_util pluginlib rclcpp rclcpp_lifecycle std_msgs tf2 tf2_ros)
  find_package(${dep} REQUIRED)
endforeach()
add_library(footbath_explore_controller SHARED src/explore_controller.cpp)
target_include_directories(footbath_explore_controller PRIVATE include)
ament_target_dependencies(footbath_explore_controller nav2_rotation_shim_controller nav2_core nav2_costmap_2d nav2_util pluginlib rclcpp rclcpp_lifecycle std_msgs tf2 tf2_ros)
pluginlib_export_plugin_description_file(nav2_core explore_controller.xml)
install(TARGETS footbath_explore_controller LIBRARY DESTINATION lib)
install(DIRECTORY config launch maps rviz DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_lint_auto REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_lint_auto_find_test_dependencies()
  add_executable(test_alignment_guard test/test_alignment_guard.cpp)
  target_include_directories(test_alignment_guard PRIVATE include)
  target_compile_options(test_alignment_guard PRIVATE -UNDEBUG)
  add_test(NAME alignment_guard COMMAND test_alignment_guard)
  ament_add_pytest_test(return_controller_profile test/test_return_controller_profile.py TIMEOUT 60)
  ament_add_pytest_test(return_controller_loading test/test_return_controller_loading.py TIMEOUT 90)
endif()
ament_package()
