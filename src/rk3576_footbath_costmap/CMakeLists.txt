cmake_minimum_required(VERSION 3.8)
project(rk3576_footbath_costmap)
find_package(ament_cmake REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_library(recovering_range_layer SHARED src/recovering_range_layer.cpp src/hazard_layer.cpp)
target_compile_features(recovering_range_layer PUBLIC cxx_std_17)
target_include_directories(recovering_range_layer PUBLIC include)
ament_target_dependencies(recovering_range_layer nav2_costmap_2d pluginlib rclcpp geometry_msgs tf2_ros sensor_msgs std_msgs)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS recovering_range_layer LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_recovery test/test_recovery.cpp)
  target_include_directories(test_recovery PRIVATE include)
  ament_add_gtest(test_hazard_layer test/test_hazard_layer.cpp)
  ament_target_dependencies(test_hazard_layer nav2_costmap_2d pluginlib rclcpp geometry_msgs tf2_ros std_msgs)
endif()
ament_package()
